232 IEnumerator Crossfade()
233 {
234 float startPresent = presentSource.volume;
235 float startPast = pastSource.volume;
236 float targetPresent = inPast ? 0f : musicVolume;
237 float targetPast = inPast ? musicVolume : 0f;
238
239 // Filters are only switched on while warping, so they can never leave the music muffled.
240 presentFilter.enabled = muffleDuringWarp;
241 pastFilter.enabled = muffleDuringWarp;
242
243 for (float t = 0f; t < crossfadeTime; t += Time.unscaledDeltaTime)
244 {
245 float k = t / crossfadeTime;
246 // Equal-power crossfade: keeps the overall loudness steady through the middle.
247 float fadeIn = Mathf.Sin(k * Mathf.PI * 0.5f);
248 float fadeOut = Mathf.Cos(k * Mathf.PI * 0.5f);
249 presentSource.volume = inPast ? startPresent * fadeOut : Mathf.Lerp(startPresent, targetPresent, fadeIn);
250 pastSource.volume = inPast ? Mathf.Lerp(startPast, targetPast, fadeIn) : startPast * fadeOut;
251
252 if (muffleDuringWarp)
253 {
254 // Dips down to muffledCutoff halfway through, then opens back up.
255 float dip = Mathf.Sin(k * Mathf.PI);
256 float cutoff = Mathf.Lerp(OpenCutoff, muffledCutoff, dip);
257 presentFilter.cutoffFrequency = cutoff;
258 pastFilter.cutoffFrequency = cutoff;
259 }
260 yield return null;
261 }
262
263 presentSource.volume = targetPresent;
264 pastSource.volume = targetPast;
265 presentFilter.cutoffFrequency = OpenCutoff;
266 pastFilter.cutoffFrequency = OpenCutoff;
267 presentFilter.enabled = false;
268 pastFilter.enabled = false;
269 fadeRoutine = null;
270
271 if (logVolumes)
272 Debug.Log($"[WarpAudio] Now in {(inPast ? "Past" : "Present")}. " +
273 $"Present vol {presentSource.volume:0.00} ({(presentSource.clip ? presentSource.clip.name : "NO CLIP")}), " +
274 $"Past vol {pastSource.volume:0.00} ({(pastSource.clip ? pastSource.clip.name : "NO CLIP")}), " +
275 $"playing: {presentSource.isPlaying}/{pastSource.isPlaying}");
276 }

300 AudioSource MakeMusicSource(string name, AudioClip clip, out AudioLowPassFilter filter)
301 {
302 var child = new GameObject(name);
303 child.transform.SetParent(transform, false);
304
305 var source = child.AddComponent<AudioSource>();
306 source.clip = clip;
307 source.loop = true;
308 source.playOnAwake = false;
309 source.volume = 0f;
310 source.spatialBlend = 0f; // plain 2D music: distance to the camera never matters
311 source.dopplerLevel = 0f;
312 source.priority = 0; // music is never cut for other sounds
313
314 filter = child.AddComponent<AudioLowPassFilter>();
315 filter.cutoffFrequency = OpenCutoff;
316 filter.enabled = false; // only switched on during a warp
317 return source;
318 }
