251 IEnumerator TypeDialogue()
252 {
253 isTyping = true;

255 dialogueText.ForceMeshUpdate();
256
257 int totalCharacters = dialogueText.textInfo.characterCount;
258
259 for (int i = 0; i < totalCharacters; i++)
260 {
261 dialogueText.maxVisibleCharacters = i + 1;
262 yield return new WaitForSeconds(typingSpeed);
263 }

265 FinishTyping();
