171 public bool PlayerTargetTimeCollision(Tilemap map)
172 {
173
174 Bounds bound = playerCollider.bounds;
175
176 //Shrink bound so standing on a floor doesn't cause swapping to fail
177 bound.Expand(-0.05f);
178 Vector3 checkFloorCollision = bound.min + new Vector3(0, feetIgnore, 0);
179
180 //Find grid cells and the bottom-left and top-right corners of the player's collider
181 Vector3Int bottomLeftCell = Vector3Int.FloorToInt(map.transform.InverseTransformPoint(checkFloorCollision));
182 Vector3Int topRightCell = Vector3Int.FloorToInt(map.transform.InverseTransformPoint(bound.max));
183
184 //Iterate through every cell the player covers and check for collision
185 for (int x = bottomLeftCell.x; x <= topRightCell.x; x++)
186 {
187
188 for (int y = bottomLeftCell.y; y <= topRightCell.y; y++)
189 {
190
191 //Check cell for a tile
192 if(map.HasTile(new Vector3Int(x, y, 0)))
193 {
194 Debug.Log("Blocked by " + map.name + " at cell (" + x + ", " + y + "), checking " + bottomLeftCell + " to " + topRightCell);
195 return true;
196
197 }
198
199 }
200
201 }
202
203 Debug.Log(map.name + ": checked cells " + bottomLeftCell + " to " + topRightCell + ", no tile found");
204 return false;
